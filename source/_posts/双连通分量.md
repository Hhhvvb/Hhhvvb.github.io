[例题: Codeforces Round 1043 (Div.3) F](https://codeforces.com/contest/2132/problem/F)

## 简述思路
- 使用tarjan算法标记出图中所有桥标记为集合A
- 找到一条从1到n的路径集合B
- 集合A与B的交集即为起点到终点所必经的点
- 最后跑一遍多源bfs预处理出所有点的答案

### 找出桥的位置
双连通分量中要注意的是如何判断当前所dfs的边不是**当前点**走向**父节点**的边, 以及当且仅当low[v] > dfn[u]时该边为桥. 这里有两种处理方式:

1. 建图时给每一条边给上编号, 例如: [0, 1]是一对相同的边的编号, [2, 3]是一对相同的编号, 查询时只需判断当前边异或上1是否等于上一条边, 如果等于则当前遍历的边是走向父节点的点, 需要continue掉.
```cpp
//建图
for (int i = 0; i < m; i ++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back({v, i << 1});
    adj[v].push_back({u, i << 1 | 1});
}

//查询时
void tarjan(int u, int last) {
    ...
    for (int i = 0; i < adj[u].size(); i ++) {
        int v = adj[u][i][0], id = adj[u][i][1];
        if ((id ^ 1) == last) continue;
        if (!dfn[v]) {
            tarjan(v, id);
            low[u] = min(low[u], low[v]);
            if (low[v] > dfn[u]) {
                is_bridge[id] = is_bridge[id ^ 1] = true;
            }
        } else {
            ...
        }
        ...
    }
    ...
}
```

2. 开一个数组用来存储每条边的两个点和一个用来存点可以到达的边的数组, 这样查询时只需判断当前边的id是否是上一条边的id即可.
```cpp
//建图
for (int i = 0; i < m; i ++) {
    int u, v;
    cin >> u >> v;
    edge[i] = {u, v};
    g[u].push_back(i);
    g[v].push_back(i);
}

//查询
void tarjan(int u, int last) {
    ...
    for (auto id : g[u]) {
        if (id == last) continue;
        int v = edge[id][0] + edge[id][1] - u;
        if (!dfn[v]) {
            tarjan(v, id);
            low[u] = min(low[u], low[v]);
            if (low[v] > dfn[u]) is_bridge[id] = true;
        } else {
            ...
        }
        ...
    }
    ...
}
```

### 找一条1-n的任意路径
这里我们当然可以再次从1开始dfs一遍找到n后标记整条路径, 但一种更巧妙的方式是在做tarjan时候找出这条路径. 我们可以定义一个全局变量`mark = 1`, 在进行tarjan前让`path[u] ^= mark`, tarjan后再让`path[u] ^= mark`, 当找到n时标记`mark = 0`. 为什么可以这样做呢?, 我们可以想象tarjan就是一个dfs过程, 只有第一次dfs到n的那一条路径中的元素是tarjan前`path[u] ^= 1`, tarjan后(已经找到n)`path[u] ^= 0`, 而未找到n的路径元素都是`path[u] ^= 1`, 找到n后的所有路径元素都是`path[u] ^= 0`, 故只有第一次遍历到n的路径上的元素是被标记为1.
代码实现如下:
```cpp
//找路径
int mark = 1;

void tarjan(int u, int last) {
    ...
    if (u == n) mark = 0;
    for (auto id : g[u]) {
        if (id == last) continue;
        int v = edge[id][0] + edge[id][1] - u;
        if (!dfn[v]) {
            path[id] ^= mark;
            tarjan(v, id);
            path[id] ^= mark;
            low[u] = min(low[u], low[v]);
            if (low[v] > dfn[u]) is_bridge[id] = true;
        } else {
            ...
        }
        ...
    }
    ...
}
```

### 初始化必经点
用数组ans来存每个点到最近必经点是哪个, dist数组为多源bfs中的距离数组.
```cpp
//初始化
vector<int> dist(n + 1, m + 1), ans(n + 1, m + 1);
queue<int> q;
for (int i = 0; i < m; i ++) {
    if (is_bridge[i] && path[i]) {
        int u = edge[i][0], v = edge[i][1];
        if (dist[u]) q.push(u);
        if (dist[v]) q.push(v);
        dist[u] = dist[v] = 0;
        ans[u] = min(ans[u], i + 1);
        ans[v] = min(ans[v], i + 1);
    }
}
```

### 跑一遍简单的多源bfs
这里就是一个很简单的多源bfs, 唯一要注意的点是依题意: 当`dist[u] == dist[v] + 1`时我们要取ans[u]的最小值
```cpp
//多源bfs
while (q.size()) {
    int u = q.front();
    q.pop();
    for (auto id : g[u]) {
        int v = edge[id][0] + edge[id][1] - u;
        if (dist[v] > dist[u] + 1) {
            dist[v] = dist[u] + 1;
            ans[v] = ans[u];
            q.push(v);
        } else if (dist[v] == dist[u] + 1) {
            ans[v] = min(ans[u], ans[v]);
        }
    }
}
```

## AC代码
```cpp
//完整代码
#include <bits/stdc++.h>

using namespace std;
using ll = long long;

void solve() {
    int n, m;
    cin >> n >> m;
    vector<array<int, 2>> edge(m);
    vector<vector<int>> g(n + 1);
    for (int i = 0; i < m; i ++) {
        int u, v;
        cin >> u >> v;
        edge[i] = {u, v};
        g[u].push_back(i);
        g[v].push_back(i);
    }
    vector<int> dfn(n + 1), low(n + 1), path(m + 1);
    vector<bool> is_bridge(m + 1);
    int timestamp = 0, mark = 1;

    function<void(int, int)> tarjan = [&](int u, int last) {
        dfn[u] = low[u] = ++ timestamp;
        if (u == n) mark = 0;
        for (auto id : g[u]) {
            if (id == last) continue;
            int v = edge[id][0] + edge[id][1] - u;
            if (!dfn[v]) {
                path[id] ^= mark;
                tarjan(v, id);
                path[id] ^= mark;
                low[u] = min(low[u], low[v]);
                if (low[v] > dfn[u]) is_bridge[id] = true;
            } else low[u] = min(low[u], dfn[v]);
        }
    };

    tarjan(1, -1);
    vector<int> dist(n + 1, m + 1), ans(n + 1, m + 1);
    queue<int> q;
    for (int i = 0; i < m; i ++) {
        if (is_bridge[i] && path[i]) {
            int u = edge[i][0], v = edge[i][1];
            if (dist[u]) q.push(u);
            if (dist[v]) q.push(v);
            dist[u] = dist[v] = 0;
            ans[u] = min(ans[u], i + 1);
            ans[v] = min(ans[v], i + 1);
        }
    }
    while (q.size()) {
        int u = q.front();
        q.pop();
        for (auto id : g[u]) {
            int v = edge[id][0] + edge[id][1] - u;
            if (dist[v] > dist[u] + 1) {
                dist[v] = dist[u] + 1;
                ans[v] = ans[u];
                q.push(v);
            } else if (dist[v] == dist[u] + 1) {
                ans[v] = min(ans[u], ans[v]);
            }
        }
    }
    int Q;
    cin >> Q;
    while (Q --) {
        int x;
        cin >> x;
        if (ans[x] != m + 1) cout << ans[x] << ' ';
        else cout << -1 << ' ';
    }
    cout << endl;
}

int main() {
    ios::sync_with_stdio(0), cin.tie(0);
    int T = 1;
    cin >> T;
    while (T --) {
        solve();
    }
}
```
