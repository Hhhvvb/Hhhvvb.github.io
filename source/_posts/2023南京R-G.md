[ICPC 2023 Nanjing R-G题](https://www.luogu.com.cn/problem/P13957)

## 简要思路
假设免费宝石为0个, 那么对n个付费物品做01背包即可.当免费宝石不止0个时, 我们可以想象最优策略应该是在付费的01背包里将**价格最贵的那个宝石免费**, 这样我们将会**剩最多的金钱**来购买未付费的宝石, 因此一种贪心思路是:将n个物品**按照价格从小到大排序**, 枚举零界点i使得前i个付费物品用01背包求出最大价值, 后n-i个物品里选出k个价值最高的宝石.

### 用小根堆来维护后n-i个物品中k个最大价值
```cpp
//示例
priority_queue<int, vector<int>, greater<int>> q;
vector<ll> suf(n + 2);
for (int i = n; i >= n - k + 1; i --) {
    q.push(a[i][1]);
    suf[i] = suf[i + 1] + a[i][1];
}
for (int i = n - k; i >= 1; i --) {
    q.push(a[i][1]);
    suf[i] = suf[i + 1] + a[i][1] - q.top();
    q.pop();
}
```

### 对n个物品做01背包
```cpp
//示例
vector<vector<ll>> dp(n + 1, vector<ll>(m + 1));
for (int i = 1; i <= n; i ++) {
    for (int j = m; j >= 0; j --) {
        dp[i][j] = dp[i - 1][j];
        if (j >= a[i][0]) {
            dp[i][j] = max(dp[i][j], dp[i - 1][j - a[i][0]] + a[i][1]);
        }
    }
}
```

### 枚举临界点i
```cpp
//示例
ll ans = suf[1];
for (int i = 1; i <= n; i ++) ans = max(ans, dp[i][m] + suf[i + 1]);
```

## AC代码
```cpp
//示例
#include <bits/stdc++.h>

using namespace std;
using ll = long long;

void solve() {
    int n, m, k;
    cin >> n >> m >> k;
    vector<array<int, 2>> a(n + 1);
    for (int i = 1; i <= n; i ++) {
        int v, w;
        cin >> v >> w;
        a[i] = {v, w};
    }
    sort(a.begin(), a.end());
    priority_queue<int, vector<int>, greater<int>> q;
    vector<ll> suf(n + 2);
    for (int i = n; i >= n - k + 1; i --) {
        q.push(a[i][1]);
        suf[i] = suf[i + 1] + a[i][1];
    }
    for (int i = n - k; i >= 1; i --) {
        q.push(a[i][1]);
        suf[i] = suf[i + 1] + a[i][1] - q.top();
        q.pop();
    }
    vector<vector<ll>> dp(n + 1, vector<ll>(m + 1));
    for (int i = 1; i <= n; i ++) {
        for (int j = m; j >= 0; j --) {
            dp[i][j] = dp[i - 1][j];
            if (j >= a[i][0]) {
                dp[i][j] = max(dp[i][j], dp[i - 1][j - a[i][0]] + a[i][1]);
            }
        }
    }
    ll ans = suf[1];
    for (int i = 1; i <= n; i ++) ans = max(ans, dp[i][m] + suf[i + 1]);
    cout << ans << endl;
}

int main() {
    ios::sync_with_stdio(0), cin.tie(0);
    int T = 1;
    // cin >> T;
    while (T --) {
        solve();
    }
}
```
